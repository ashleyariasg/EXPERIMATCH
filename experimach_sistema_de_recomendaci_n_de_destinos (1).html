Script SQL DDL (bd_lugares) 


-- ============================================================
-- SCRIPT SQL DE CREACIÓN DE BASE DE DATOS Y TABLAS
-- ============================================================
CREATE DATABASE IF NOT EXISTS bd_lugares;
USE bd_lugares;

CREATE TABLE CIUDAD (
    id_ciudad INT AUTO_INCREMENT PRIMARY KEY,
    nombre_ciudad VARCHAR(100) NOT NULL
);

CREATE TABLE ZONA (
    id_zona INT AUTO_INCREMENT PRIMARY KEY,
    nombre_zona VARCHAR(100) NOT NULL,
    id_ciudad INT NOT NULL,
    CONSTRAINT fk_zona_ciudad FOREIGN KEY (id_ciudad) REFERENCES CIUDAD(id_ciudad) ON DELETE CASCADE
);

CREATE TABLE GENERO (
    id_genero INT AUTO_INCREMENT PRIMARY KEY,
    tipo_genero VARCHAR(50) NOT NULL
);

CREATE TABLE USUARIO (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    nombre_usuario VARCHAR(100) NOT NULL,
    id_genero INT NOT NULL,
    fecha_nacimiento DATE,
    CONSTRAINT fk_usuario_genero FOREIGN KEY (id_genero) REFERENCES GENERO(id_genero)
);

CREATE TABLE LUGARES (
    id_lugares INT AUTO_INCREMENT PRIMARY KEY,
    id_ciudad INT NOT NULL,
    nombre_lugar VARCHAR(100) NOT NULL,
    descripcion VARCHAR(300),
    edad_minima INT DEFAULT 0,
    calle VARCHAR(50),
    carrera VARCHAR(50),
    numero INT,
    CONSTRAINT fk_lugares_ciudad FOREIGN KEY (id_ciudad) REFERENCES CIUDAD(id_ciudad) ON DELETE CASCADE
);

CREATE TABLE TIPO_LUGAR (
    id_tipo_lugar INT AUTO_INCREMENT PRIMARY KEY,
    tipo_lugar VARCHAR(100) NOT NULL,
    id_lugares INT NOT NULL,
    CONSTRAINT fk_tipo_lugar_lugares FOREIGN KEY (id_lugares) REFERENCES LUGARES(id_lugares) ON DELETE CASCADE
);

CREATE TABLE USUARIO_LUGARES (
    usuario_lugares INT AUTO_INCREMENT PRIMARY KEY,
    id_lugares INT NOT NULL,
    id_usuario INT NOT NULL,
    fecha_visita DATE,
    CONSTRAINT fk_usuariolugares_lugares FOREIGN KEY (id_lugares) REFERENCES LUGARES(id_lugares) ON DELETE CASCADE,
    CONSTRAINT fk_usuariolugares_usuario FOREIGN KEY (id_usuario) REFERENCES USUARIO(id_usuario) ON DELETE CASCADE
);

CREATE TABLE ETIQUETA (
    id_etiqueta INT AUTO_INCREMENT PRIMARY KEY,
    nombre_etiqueta VARCHAR(50) NOT NULL
);

CREATE TABLE LUGAR_ETIQUETA (
    id_lugar_etiqueta INT AUTO_INCREMENT PRIMARY KEY,
    id_etiqueta INT NOT NULL,
    id_lugar INT NOT NULL,
    CONSTRAINT fk_lugaretiqueta_etiqueta FOREIGN KEY (id_etiqueta) REFERENCES ETIQUETA(id_etiqueta) ON DELETE CASCADE,
    CONSTRAINT fk_lugaretiqueta_lugar FOREIGN KEY (id_lugar) REFERENCES LUGARES(id_lugares) ON DELETE CASCADE
);

CREATE TABLE RESENA (
    id_resena INT AUTO_INCREMENT PRIMARY KEY,
    fecha DATE,
    calificacion INT,
    comentario VARCHAR(300),
    id_lugares INT NOT NULL,
    id_usuario INT NOT NULL,
    CONSTRAINT fk_resena_lugares FOREIGN KEY (id_lugares) REFERENCES LUGARES(id_lugares) ON DELETE CASCADE,
    CONSTRAINT fk_resena_usuario FOREIGN KEY (id_usuario) REFERENCES USUARIO(id_usuario) ON DELETE CASCADE
);
